ScrapingBee is a scraping API that handles Cloudflare bypass

Sign up at: scrapingbee.com (free, no credit card)
Add to Replit Secrets: SCRAPINGBEE_API_KEY=your_key_here

Create /lib/scrapers/scrapingbee.ts:

const SCRAPINGBEE_URL = 'https://app.scrapingbee.com/api/v1/';

export async function fetchWithScrapingBee(
  targetUrl: string,
  options: { render_js?: boolean; premium_proxy?: boolean } = {}
): Promise<string | null> {
  const apiKey = process.env.SCRAPINGBEE_API_KEY;
  if (!apiKey) return null;
  
  try {
    const params = new URLSearchParams({
      api_key: apiKey,
      url: targetUrl,
      render_js: String(options.render_js ?? true),
      premium_proxy: String(options.premium_proxy ?? false),
      country_code: 'nz',
      block_ads: 'true',
      wait: '2000',
    });
    
    const response = await fetch(
      `${SCRAPINGBEE_URL}?${params.toString()}`,
      { signal: AbortSignal.timeout(30000) }
    );
    
    if (!response.ok) {
      console.warn(`ScrapingBee error: ${response.status}`);
      return null;
    }
    
    return await response.text();
  } catch (err) {
    console.warn('ScrapingBee fetch failed:', err);
    return null;
  }
}

export async function scrapeHougarden(
  lat: number, lng: number, address: string
): Promise<HougardenData> {
  
  // ATTEMPT 1: Playwright stealth
  try {
    const result = await scrapeHougardenPlaywright(lat, lng, address);
    if (result.zone_code || result.overlays.length > 0) {
      console.log('Hougarden: stealth playwright succeeded');
      return result;
    }
  } catch (err) {
    console.warn('Hougarden stealth attempt failed:', err);
  }
  
  // ATTEMPT 2: ScrapingBee
  try {
    const result = await scrapeHougardenViaBee(address);
    if (result) {
      console.log('Hougarden: ScrapingBee succeeded');
      return result;
    }
  } catch (err) {
    console.warn('Hougarden ScrapingBee attempt failed:', err);
  }
  
  // FALLBACK: return empty with flag
  console.warn('Hougarden: all attempts failed, returning empty');
  return emptyHougardenData();
}

async function scrapeHougardenViaBee(
  address: string
): Promise<HougardenData | null> {
  const { fetchWithScrapingBee } = await import('./scrapingbee');
  
  const searchUrl = `https://www.hougarden.com/search/` +
    `#ac=1&address=${encodeURIComponent(address)}`;
  
  const html = await fetchWithScrapingBee(searchUrl, { 
    render_js: true,
    premium_proxy: false
  });
  
  if (!html) return null;
  
  // Parse the returned HTML with cheerio
  // (npm install cheerio)
  const cheerio = await import('cheerio');
  const $ = cheerio.load(html);
  
  // Extract using same selectors as Playwright version
  // but using cheerio's jQuery-like API on the rendered HTML
  const cvText = $('[class*="cv"], [class*="capital-value"]').first().text();
  const zoneText = $('[class*="zone"]').first().text();
  // ... extract other fields similarly
  
  return parseHougardenFields({ cvText, zoneText });
}
