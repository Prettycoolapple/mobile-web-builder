Step 1 — Detect development intent from report:

export async function detectDevelopmentIntent(
  report: FeasibilityReport,
  conversationHistory: Message[]
): Promise<{
  shouldRecommend: boolean,
  intentType: 'subdivision' | 'newbuild' | 'renovation' | 'none',
  confidence: number,
  reason: string,
}> {
  
  // Rule-based fast check first (no API call needed)
  const signals = {
    isSubdividable: (report.lots?.lots ?? 0) >= 2,
    isDevelopmentZone: ['THAB','MHU','MHS','MIX'].includes(
      report.merged?.zone_code ?? ''
    ),
    isLargeLand: (report.merged?.land_area_sqm ?? 0) > 500,
    hasGoodEase: (report.scores?.ease ?? 0) >= 2.5,
    hasPositiveROI: report.scenarios?.some(s => s.viable) ?? false,
  };
  
  const signalCount = Object.values(signals).filter(Boolean).length;
  
  // Check conversation for explicit developer intent
  const recentMessages = conversationHistory
    .slice(-6)
    .map(m => m.content)
    .join(' ')
    .toLowerCase();
  
  const developerKeywords = [
    'build', 'develop', 'subdivide', 'architect', 
    'builder', 'engineer', 'construction', 'consent',
    'townhouse', 'units', 'new build',
  ];
  
  const hasKeywordIntent = developerKeywords.some(
    kw => recentMessages.includes(kw)
  );
  
  // Strong signal: 3+ property signals OR keyword + 2 signals
  if (signalCount >= 3 || (hasKeywordIntent && signalCount >= 2)) {
    
    // Confirm with Claude for nuance
    const claudeCheck = await anthropic.messages.create({
      model: 'claude-sonnet-4-5',
      max_tokens: 150,
      messages: [{
        role: 'user',
        content: `A user just analysed a NZ property with these details:
        Zone: ${report.merged?.zone_code}
        Potential lots: ${report.lots?.lots}
        Land area: ${report.merged?.land_area_sqm}m²
        Ease score: ${report.scores?.ease}/5
        ROI viable: ${report.scenarios?.some(s => s.viable)}
        
        Recent conversation: "${recentMessages.slice(-200)}"
        
        Should we recommend a development service provider 
        (builder, architect, or project manager)?
        
        Reply with ONLY valid JSON:
        {"recommend": true/false, "type": "subdivision|newbuild|renovation|none", "reason": "one sentence"}`
      }]
    });
    
    try {
      const result = JSON.parse(claudeCheck.content[0].text);
      return {
        shouldRecommend: result.recommend,
        intentType: result.type,
        confidence: signalCount / 5,
        reason: result.reason,
      };
    } catch {
      // Claude parse failed — use rule-based result
      return {
        shouldRecommend: true,
        intentType: signals.isSubdividable ? 'subdivision' : 'newbuild',
        confidence: signalCount / 5,
        reason: 'Property shows strong development potential',
      };
    }
  }
  
  return { shouldRecommend: false, intentType: 'none', 
           confidence: 0, reason: '' };
}

export async function selectServiceProvider(
  intentType: string
): Promise<ServiceProvider | null> {
  
  // Fetch top 3 providers by recommendation count
  // (recommendation_count column on profiles table)
  const { data: providers } = await supabase
    .from('profiles')
    .select(
      'id, full_name, business_name, specialties, ' +
      'languages, recommendation_count, avatar_url, ' +
      'contact_mobile, bio'
    )
    .eq('account_type', 'service_provider')
    .eq('is_active', true)
    .order('recommendation_count', { ascending: false })
    .limit(3);
  
  if (!providers?.length) return null;
  
  // For now: randomly select from top 3
  // Future: match by location, specialty, language
  const selected = providers[
    Math.floor(Math.random() * providers.length)
  ];
  
  return selected as ServiceProvider;
}

export async function connectWithProvider(
  generalUserId: string,
  providerId: string,
  propertyAddress: string,
  analysisReportId: string,
): Promise<{ conversationId: string }> {
  
  // 1. Check if conversation already exists
  const { data: existing } = await supabase
    .from('conversations')
    .select('id')
    .eq('general_user_id', generalUserId)
    .eq('provider_id', providerId)
    .eq('property_address', propertyAddress)
    .single();
  
  if (existing) return { conversationId: existing.id };
  
  // 2. Create new conversation
  const { data: conversation } = await supabase
    .from('conversations')
    .insert({
      general_user_id: generalUserId,
      provider_id: providerId,
      property_address: propertyAddress,
      analysis_report_id: analysisReportId,
    })
    .select('id')
    .single();
  
  // 3. Send automated opening message from AI
  const openingMessage = 
    `Hi! I've been connected with you regarding ` +
    `${propertyAddress}. The AI analysis shows this ` +
    `property has development potential — ` +
    `I'd love to discuss further.`;
  
  await supabase.from('messages').insert({
    conversation_id: conversation.id,
    sender_id: generalUserId,
    content: openingMessage,
    message_type: 'text',
  });
  
  // 4. Increment provider recommendation count
  await supabase.rpc('increment_recommendation_count', {
    provider_id: providerId
  });
  
  // 5. Send push notification to provider (if token exists)
  await notifyProvider(providerId, generalUserId, propertyAddress);
  
  return { conversationId: conversation.id };
}

import * as Notifications from 'expo-notifications';
import Constants from 'expo-constants';

// On app start, register for push notifications
export async function registerForPushNotifications(
  userId: string
): Promise<void> {
  const { status } = await Notifications.requestPermissionsAsync();
  if (status !== 'granted') return;
  
  const token = await Notifications.getExpoPushTokenAsync({
    projectId: Constants.expoConfig?.extra?.eas?.projectId,
  });
  
  // Save token to profiles table
  await supabase
    .from('profiles')
    .update({ push_token: token.data })
    .eq('id', userId);
}

// Add push_token column to profiles:
// ALTER TABLE public.profiles 
// ADD COLUMN IF NOT EXISTS push_token text;

// Notify provider when user connects:
async function notifyProvider(
  providerId: string,
  generalUserId: string,
  propertyAddress: string,
): Promise<void> {
  
  const { data: provider } = await supabase
    .from('profiles')
    .select('push_token, full_name')
    .eq('id', providerId)
    .single();
  
  if (!provider?.push_token) return;
  
  // Send via Expo Push API
  await fetch('https://exp.host/--/api/v2/push/send', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      to: provider.push_token,
      title: 'New connection request',
      body: `A user wants to discuss ${propertyAddress}`,
      data: { type: 'new_connection', address: propertyAddress },
    }),
  });
}
