STEP 6 — Create /api/searches GET endpoint if missing.

// /app/api/searches/route.ts
export async function GET(request: Request) {
  const supabase = createServerClient(...);
  const { data: { user } } = await supabase.auth.getUser();
  
  if (!user) return Response.json({ error: 'Unauthorized' }, 
    { status: 401 });
  
  const { data, error } = await supabase
    .from('searches')
    .select('id, address, created_at, result_json')
    .eq('user_id', user.id)
    .order('created_at', { ascending: false })
    .limit(20);
  
  if (error) return Response.json({ error }, { status: 500 });
  
  // Return lightweight version — don't send full JSON to list
  const summary = data.map(s => ({
    id: s.id,
    address: s.address,
    created_at: s.created_at,
    composite_score: s.result_json?.scores?.composite ?? null,
    zone: s.result_json?.merged?.zone_code ?? null,
  }));
  
  return Response.json(summary);
}
