3. COST ESTIMATOR
/lib/cost-estimator.ts
════════════════════════════════

Function: estimateCosts(
  data: MergedPropertyData,
  units: number
): CostBreakdown

INPUT DEPENDENCIES (from merged data):
- cv_nzd: land value proxy
- contour: 'flat'|'gentle'|'moderate'|'steep'
- asbestos_risk: 'low'|'high'|'unknown'
- infrastructure: InfrastructureItem[]
- has_existing_dwelling: boolean (true if build_year exists)

CALCULATIONS:

a) LAND: land_cv_nzd = merged.cv_nzd ?? 0

b) DEMOLITION (only if has_existing_dwelling):
   low risk:     { low: 15000,  high: 30000 }
   high risk:    { low: 35000,  high: 80000 }
   unknown risk: { low: 20000,  high: 60000 }
   no dwelling:  { low: 0,      high: 0     }

c) RETAINING WALLS (from contour):
   flat:     { low: 0,      high: 0      }
   gentle:   { low: 10000,  high: 30000  }
   moderate: { low: 50000,  high: 150000 }
   steep:    { low: 200000, high: 400000 }

d) SERVICE CONNECTIONS (sum infrastructure items):
   Sum all InfrastructureItem cost_estimate_low/high

e) CONSTRUCTION:
   floor_area_per_unit = 120  // m², NZ townhouse standard
   rate_low  = 2800  // $/m²
   rate_high = 3500  // $/m²
   
   // Steep site premium
   if contour === 'steep':
     rate_low  *= 1.18
     rate_high *= 1.18
   else if contour === 'moderate':
     rate_low  *= 1.08
     rate_high *= 1.08
   
   construction_low  = rate_low  × floor_area_per_unit × units
   construction_high = rate_high × floor_area_per_unit × units

f) CONSENTS & PROFESSIONAL FEES:
   consents_low  = construction_low  × 0.13
   consents_high = construction_high × 0.16

g) FINANCE / HOLDING COSTS:
   // 7.5% p.a. on land + 50% of construction midpoint
   construction_mid = (construction_low + construction_high) / 2
   loan_base = (cv_nzd ?? 0) + (construction_mid * 0.5)
   finance_low  = loan_base × 0.075 × 1.5  // 18 months
   finance_high = loan_base × 0.075 × 2.5  // 30 months

h) CONTINGENCY:
   subtotal_low  = demo_low  + retaining_low  + services_low  + 
                   construction_low  + consents_low  + finance_low
   subtotal_high = demo_high + retaining_high + services_high + 
                   construction_high + consents_high + finance_high
   contingency_low  = subtotal_low  × 0.08
   contingency_high = subtotal_high × 0.12

i) TOTALS:
   total_low  = cv_nzd + subtotal_low  + contingency_low
   total_high = cv_nzd + subtotal_high + contingency_high
   cost_per_unit_avg = ((total_low + total_high) / 2) / units

5. ROI CALCULATOR
/lib/roi-calculator.ts
════════════════════════════════

Function: calculateROIScenarios(
  costs: CostBreakdown,
  avg_sale_price_per_unit: number,
  units: number
): ROIScenario[]

gdv = avg_sale_price_per_unit × units
total_cost_mid = (costs.total_low + costs.total_high) / 2

For years in [2, 3, 4]:
  gross_profit = gdv - total_cost_mid
  roi_percent  = (gross_profit / total_cost_mid) × 100
  
  // Annualised using compound formula
  annualised_roi_percent = 
    (Math.pow(1 + roi_percent / 100, 1 / years) - 1) × 100
  
  viable = gross_profit > 0

Round all percentages to 1 decimal place.
Round all dollar values to nearest 1000.

Function: scoreProperty(
  merged: MergedPropertyData,
  costs: CostBreakdown,
  scenarios: ROIScenario[],
  lots: number
): ScoringResult

— EASE SCORE (start at 5.0, deduct) —

const easeDeductions: Array<{ condition: boolean, points: number, reason: string }> = [
  {
    condition: merged.zone_code === 'SHZ',
    points: 1.5,
    reason: 'Single House Zone — subdivision heavily restricted'
  },
  {
    condition: merged.zone_code === 'LSZ' || merged.zone_code === 'RUR',
    points: 2.0,
    reason: 'Large Lot / Rural zone — development very limited'
  },
  {
    condition: hasOverlay(merged, 'heritage'),
    points: 1.5,
    reason: 'Heritage overlay — demolition may require resource consent'
  },
  {
    condition: hasOverlay(merged, 'flood'),
    points: 1.2,
    reason: 'Flood overlay — engineering and consent complexity'
  },
  {
    condition: hasOverlay(merged, 'tree'),
    points: 0.5,
    reason: 'Notable tree overlay — design constraints apply'
  },
  {
    condition: hasOverlay(merged, 'viewshaft'),
    points: 0.5,
    reason: 'Volcanic viewshaft — height restrictions apply'
  },
  {
    condition: hasOverlay(merged, 'coastal'),
    points: 0.8,
    reason: 'Coastal protection overlay — additional consenting required'
  },
  {
    condition: merged.contour === 'steep',
    points: 0.8,
    reason: 'Steep terrain — significant earthworks required'
  },
  {
    condition: merged.contour === 'moderate',
    points: 0.4,
    reason: 'Moderate slope — some retaining wall work expected'
  },
  {
    condition: merged.asbestos_risk === 'high',
    points: 0.4,
    reason: 'Probable asbestos — specialist demolition required'
  },
  {
    condition: hasNeighbourInfrastructure(merged),
    points: 0.4,
    reason: 'Service infrastructure on neighbouring land — easement needed'
  },
  {
    condition: lots <= 1,
    points: 0.5,
    reason: 'Land area limits subdivision to single dwelling'
  },
];

Apply all matching deductions, collect reasons.
ease = Math.max(0.5, 5.0 - total_deductions)
Round to nearest 0.5.

Helper: hasOverlay(merged, keyword) — checks if any overlay 
name contains the keyword (case-insensitive) and status 
is 'restricted' or 'moderate'.

Helper: hasNeighbourInfrastructure(merged) — returns true 
if any InfrastructureItem has location === 'neighbour'.

— COST SCORE (5 = low cost, 1 = very high) —

costPerUnit = costs.cost_per_unit_avg

const costBrackets = [
  { max: 400000,  score: 5.0, reason: 'Excellent cost efficiency per unit' },
  { max: 550000,  score: 4.0, reason: 'Good cost per unit for NZ market' },
  { max: 700000,  score: 3.0, reason: 'Moderate cost — market viable' },
  { max: 900000,  score: 2.0, reason: 'High cost per unit — margin is thin' },
  { max: 1200000, score: 1.0, reason: 'Very high cost — ROI challenging' },
  { max: Infinity,score: 0.5, reason: 'Extreme cost — feasibility doubtful' },
];

Find first bracket where costPerUnit <= bracket.max.
cost = bracket.score
cost_reasons = [bracket.reason, 
  `Cost per unit: $${formatNZD(costPerUnit)}`]

— ROI SCORE —

bestScenario = scenarios with highest roi_percent.

const roiBrackets = [
  { min: 35,  score: 5.0, reason: 'Exceptional return — strong development opportunity' },
  { min: 25,  score: 4.0, reason: 'Strong return — well above typical NZ threshold' },
  { min: 15,  score: 3.0, reason: 'Solid return — meets typical developer hurdle rate' },
  { min: 8,   score: 2.0, reason: 'Marginal return — viable but sensitive to cost overruns' },
  { min: 0,   score: 1.0, reason: 'Low return — high risk of negative outcome' },
  { min: -Infinity, score: 0.5, reason: 'Negative return — not viable at current market values' },
];

Find first bracket where bestScenario.roi_percent >= bracket.min.
roi = bracket.score
roi_reasons = [bracket.reason,
  `Best case: ${bestScenario.roi_percent.toFixed(1)}% ROI ` +
  `over ${bestScenario.years} years`]

— COMPOSITE SCORE —
composite = (ease × 0.30) + (cost × 0.30) + (roi × 0.40)
Round to 1 decimal place.

7. ASBESTOS CLASSIFIER
/lib/asbestos.ts
════════════════════════════════

Function: classifyAsbestos(build_year: number | null): {
  risk: 'low' | 'high' | 'unknown',
  notes: string,
  worksafe_required: boolean
}

build_year === null → 'unknown':
  notes: 'Build year unknown. A pre-demolition asbestos 
  survey is recommended before any demolition works, as 
  required under WorkSafe NZ guidelines.'
  worksafe_required: true

build_year <= 1940 → 'low':
  notes: 'Pre-1940 construction. Asbestos cement products 
  were not in widespread use in NZ at this time.'
  worksafe_required: false

1940 < build_year <= 1990 → 'high':
  notes: 'Built during the peak NZ asbestos use period 
  (1940–1990). Buildings of this era commonly contain 
  asbestos cement cladding (fibrolite), textured ceiling 
  coatings, vinyl floor tiles, and pipe lagging. A 
  licensed asbestos assessor is legally required before 
  demolition under the Health and Safety at Work 
  (Asbestos) Regulations 2016.'
  worksafe_required: true

build_year > 1990 → 'low':
  notes: 'Post-1990 construction. Asbestos was largely 
  phased out of NZ building products by this time. 
  Low risk.'
  worksafe_required: false

  merged.land_area_sqm ?? 400,
  merged.zone_code
);

// 2. Asbestos

// 3. Costs

// 4. Comparables
const { comparables, avg_sale_price, data_quality } = 
  await getComparables(suburb, merged.zone_code, lat, lng);

// 5. ROI
const scenarios = calculateROIScenarios(
  costs, avg_sale_price, lotResult.lots
);

// 6. Scores
const scores = scoreProperty(merged, costs, scenarios, lotResult.lots);

// 7. Add to pipeline result
return {
  ...existingPipelineResult,
  merged,
  lots: lotResult,
  asbestos,
  costs,
  comparables,
  comparables_quality: data_quality,
  scenarios,
  scores,
};

export function formatNZD(amount: number): string {
  return new Intl.NumberFormat('en-NZ', {
    minimumFractionDigits: 0,
    maximumFractionDigits: 0,
  }).format(Math.round(amount));
}
