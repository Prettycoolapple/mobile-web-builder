async function fetchElevationFree(
  lat: number, lng: number
): Promise<ContourResult> {
  
  // Sample 9 points in a grid around the property
  const d = 0.00025; // ~25m offset
  const points = [
    [lat-d, lng-d], [lat-d, lng], [lat-d, lng+d],
    [lat,   lng-d], [lat,   lng], [lat,   lng+d],
    [lat+d, lng-d], [lat+d, lng], [lat+d, lng+d],
  ];
  
  const locationStr = points
    .map(([la, ln]) => `${la},${ln}`)
    .join('|');
  
  // SRTM 30m dataset — covers all of NZ, free, no key
  const url = `https://api.opentopodata.org/v1/srtm30m` +
    `?locations=${locationStr}`;
  
  const res = await fetch(url, {
    signal: AbortSignal.timeout(8000)
  });
  const data = await res.json();
  
  if (data.status !== 'OK') throw new Error('Elevation fetch failed');
  
  const elevations = data.results.map((r: any) => r.elevation);
  const center = elevations[4];
  
  // Max elevation change across the 50m grid
  const maxDiff = Math.max(
    ...elevations.map((e: number) => Math.abs(e - center))
  );
  
  // Steepest slope — using 35m diagonal distance
  const slopeDeg = Math.atan(maxDiff / 35) * (180 / Math.PI);
  
  const classification =
    slopeDeg < 5  ? 'flat'     :
    slopeDeg < 12 ? 'gentle'   :
    slopeDeg < 22 ? 'moderate' : 'steep';
  
  return {
    slope_degrees: Math.round(slopeDeg * 10) / 10,
    classification,
    data_source: 'Open-Topo-Data (SRTM 30m)',
    elevation_center_m: Math.round(center),
  };
}

// If opentopodata fails, try Open-Elevation as backup:
async function fetchElevationBackup(
  lat: number, lng: number
): Promise<number[]> {
  const d = 0.00025;
  const points = [
    {latitude: lat-d, longitude: lng-d},
    {latitude: lat,   longitude: lng  },
    {latitude: lat+d, longitude: lng+d},
  ];
  const res = await fetch('https://api.open-elevation.com/api/v1/lookup', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({ locations: points }),
    signal: AbortSignal.timeout(8000),
  });
  const data = await res.json();
  return data.results.map((r: any) => r.elevation);
}

// Master elevation fetcher — tries both, no key needed
export async function getContour(
  lat: number, lng: number
): Promise<ContourResult> {
  try {
    return await fetchElevationFree(lat, lng);
  } catch (e1) {
    console.warn('OpenTopoData failed, trying backup:', e1);
    try {
      const elevs = await fetchElevationBackup(lat, lng);
      const diff = Math.abs(elevs[2] - elevs[0]);
      const slopeDeg = Math.atan(diff / 50) * (180 / Math.PI);
      return {
        slope_degrees: Math.round(slopeDeg * 10) / 10,
        classification: slopeDeg < 5 ? 'flat' :
                        slopeDeg < 12 ? 'gentle' :
                        slopeDeg < 22 ? 'moderate' : 'steep',
        data_source: 'Open-Elevation API',
      };
    } catch (e2) {
      console.error('Both elevation APIs failed:', e2);
      // Only now do we give up
      return { slope_degrees: null, classification: null, 
               data_source: 'unavailable' };
    }
  }
}

    .toLowerCase()

  const suburbSlug = suburb.toLowerCase().replace(/\s+/g, '-');

  const cvPatterns = [
    /capital\s*value[^$]*\$([\d,]+)/gi,
    /cv[:\s]+\$([\d,]+)/gi,
    /"cv"[:\s]+"?\$([\d,]+)/gi,
    /rateable\s*value[^$]*\$([\d,]+)/gi,
    // homes.co.nz specific data attributes
    /data-cv="([\d]+)"/i,
    /"capitalValue"\s*:\s*([\d]+)/i,
    /"cv"\s*:\s*([\d]+)/i,
  ];
  
  for (const pattern of cvPatterns) {
    const match = pattern.exec(html);

      if (value > 100000 && value < 50000000) { // sanity check
        console.log(`[CV] homes.co.nz returned: $${value}`);
        return value;
      }
    }
  }
  return null;

async function extractCVFromHougardenPage(page: Page): 
Promise<number | null> {
  
  // Method 1: Look for CV in all text content
  const bodyText = await page.textContent('body') ?? '';
  
  const patterns = [
    /CV[:\s]+\$?([\d,]+)/g,
    /Capital Value[:\s]+\$?([\d,]+)/gi,
    /Rateable Value[:\s]+\$?([\d,]+)/gi,
    /\$?([\d,]+)\s*CV\b/g,
  ];
  
  for (const pattern of patterns) {
    let match;
    while ((match = pattern.exec(bodyText)) !== null) {
      const value = parseInt(match[1].replace(/,/g, ''));
      if (value > 200000 && value < 30000000) {
        return value;
      }
    }
  }
  
  // Method 2: Look for specific Hougarden data elements
  const selectors = [
    '[class*="cv-value"]',
    '[class*="capital-value"]',
    '[data-field="cv"]',
    '[class*="rating-valuation"]',
    'td:has-text("CV") + td',
    'dt:has-text("Capital Value") + dd',
    'dt:has-text("CV") + dd',
  ];
  
  for (const selector of selectors) {
    try {
      const text = await page.$eval(selector, 
        (el: Element) => el.textContent?.trim()
      );
      if (text) {
        const value = parseInt(text.replace(/[$,]/g, ''));
        if (value > 200000) return value;
      }
    } catch { continue; }
  }
  
  // Method 3: Look in page's __NEXT_DATA__ or window state
  const jsonData = await page.evaluate(() => {
    const scripts = Array.from(
      document.querySelectorAll('script[type="application/json"]')
    );
    return scripts.map(s => s.textContent).join(' ');
  });
  
  const jsonCVMatch = /"capitalValue"\s*:\s*(\d+)/.exec(jsonData);
  if (jsonCVMatch) return parseInt(jsonCVMatch[1]);
  
  return null;
}

async function fetchLINZParcelByCoord(

async function queryInfrastructureLayer(
  layerUrl: string,
  lat: number, lng: number,
  radiusMetres: number = 100
): Promise<any[]> {
  
  // Convert lat/lng to the map extent for the query
  const offset = radiusMetres / 111320; // degrees per metre
  
  const params = new URLSearchParams({
    geometry: `${lng},${lat}`,
    geometryType: 'esriGeometryPoint',
    inSR: '4326',
    spatialRel: 'esriSpatialRelIntersects',
    distance: String(radiusMetres),
    units: 'esriSRUnit_Meter',
    outFields: '*',
    returnGeometry: 'true',
    outSR: '4326',
    f: 'json',
  });
  
  const res = await fetch(`${layerUrl}/query?${params}`, {
    signal: AbortSignal.timeout(10000)
  });
  
  if (!res.ok) return [];
  const data = await res.json();
  return data.features ?? [];
}

async function fetchInfrastructure(
  lat: number, lng: number,
  parcelBoundaryWKT?: string  // from LINZ if available
): Promise<InfrastructureItem[]> {
  
  const results = await Promise.allSettled(
    Object.entries(INFRA_LAYERS).map(async ([type, url]) => {
      const features = await queryInfrastructureLayer(url, lat, lng, 100);
      
      if (!features.length) return { 
        service: type, location: 'unknown', features: [] 
      };
      
      // Calculate distance from property center to nearest pipe
      const distances = features.map(f => {
        // Feature geometry is a polyline or point
        // Calculate distance from (lat,lng) to nearest point on geometry
        return calculateDistance(lat, lng, f.geometry);
      });
      
      const minDistance = Math.min(...distances);
      
      let location: string;
      if (minDistance < 5)       location = 'on-parcel';
      else if (minDistance < 15) location = 'boundary';
      else if (minDistance < 40) location = 'neighbour';
      else                       location = 'public-land';
      
      return { service: type, location, min_distance_m: minDistance };
    })
  );
  
  return results.map((r, i) => {
    const type = Object.keys(INFRA_LAYERS)[i];
    const serviceNames: Record<string, string> = {
      wastewater: 'Wastewater',
      stormwater: 'Stormwater', 
      water: 'Water supply',
    };
    
    if (r.status === 'fulfilled' && r.value.location !== 'unknown') {
      return {
        service: serviceNames[type],
        location: r.value.location,
        data_source: 'Auckland Council GIS',
        min_distance_m: r.value.min_distance_m,
        ...getCostEstimate(r.value.location),
      };
    }
    
    return {
      service: serviceNames[type],
      location: 'unknown',
      data_source: 'unavailable — check Auckland Council GIS',
      cost_estimate_low: null,
      cost_estimate_high: null,
    };
  });
}

// Helper: calculate distance from point to GIS geometry
function calculateDistance(
  lat: number, lng: number, 
  geometry: any
): number {
  // For polyline geometry, check distance to each segment
  // Use simple Euclidean distance in degrees × 111320 m/degree
  if (geometry?.paths) {
    let minDist = Infinity;
    for (const path of geometry.paths) {
      for (const [gLng, gLat] of path) {
        const dist = Math.sqrt(
          Math.pow((lat - gLat) * 111320, 2) +
          Math.pow((lng - gLng) * 111320 * Math.cos(lat * Math.PI/180), 2)
        );
        minDist = Math.min(minDist, dist);
      }
    }
    return minDist;
  }
  if (geometry?.x) { // point geometry
    return Math.sqrt(
      Math.pow((lat - geometry.y) * 111320, 2) +
      Math.pow((lng - geometry.x) * 111320 * Math.cos(lat * Math.PI/180), 2)
    );
  }
  return Infinity;
}
